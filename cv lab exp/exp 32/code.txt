import cv2
import numpy as np
from google.colab.patches import cv2_imshow

def create_image(width, height):

    # Create white image
    img = np.ones((height, width, 3), dtype=np.uint8) * 255

    # Calculate box size (1/10th of image)
    box_w = width // 10
    box_h = height // 10

    # Create four colored boxes
    img[0:box_h, 0:box_w] = (0, 0, 0)                  # Black
    img[0:box_h, width-box_w:width] = (255, 0, 0)      # Blue
    img[height-box_h:height, 0:box_w] = (0, 255, 0)   # Green
    img[height-box_h:height, width-box_w:width] = (0, 0, 255) # Red

    # Display image
    cv2_imshow(img)

    # Save image
    cv2.imwrite("colored_boxes.jpg", img)


# Get image size from user
width = int(input("Enter image width: "))
height = int(input("Enter image height: "))

create_image(width, height)
